2 Distributed Lock

Vấn đề: file lock hiện tại chỉ an toàn trong 1 process/1 máy.

2.1 Hiện trạng (để hiểu)

// runtime/core/fs-store.mjs — POSIX file lock
export async function withLock(store, name, fn, options = {}) {
  const lock = path.join(store.runtimeDir, 'locks', `${name}.lock`);
  while (true) {
    try {
      await writeFile(lock, String(process.pid), { flag: 'wx' });
      break;
    } catch {
      // retry or timeout
    }
  }
  try { return await fn(); } finally { await rm(lock); }
}
⚠️ Vấn đề

Dùng flag: 'wx' (O_CREAT | O_EXCL). Chỉ atomic trong 1 filesystem. Không an toàn cho: cluster NFS, multi-pod K8s, network filesystem có cache không nhất quán.

2.2 Giải pháp A: SQLite WAL mode (đơn giản nhất, 1 máy)

// runtime/core/sqlite-store.mjs (mới)
import Database from 'better-sqlite3';

export function createSqliteStore(root = process.cwd()) {
  const dbPath = path.join(root, '.omc', 'runtime', 'store.db');
  const db = new Database(dbPath);

  // WAL = Write-Ahead Logging: nhiều reader + 1 writer, không block nhau
  db.pragma('journal_mode = WAL');
  db.pragma('synchronous = NORMAL');   // an toàn hơn FULL, nhanh hơn OFF
  db.pragma('foreign_keys = ON');

  db.exec(`
    CREATE TABLE IF NOT EXISTS collections (
      kind TEXT PRIMARY KEY,
      schema_version INTEGER NOT NULL,
      revision INTEGER NOT NULL DEFAULT 0,
      updated_at TEXT NOT NULL
    );
    CREATE TABLE IF NOT EXISTS items (
      id TEXT NOT NULL,
      kind TEXT NOT NULL,
      data JSON NOT NULL,
      created_at TEXT NOT NULL,
      updated_at TEXT NOT NULL,
      PRIMARY KEY (kind, id)
    );
  `);
  return { db, dbPath };
}

export async function withLock(store, name, fn) {
  // SQLite tự lock qua transactions, không cần file lock ngoài
  return store.db.transaction(() => fn()).immediate();
}

2.3 Giải pháp B: Redis Redlock (multi-node)

// runtime/core/redlock.mjs (mới)
import Redis from 'ioredis';
import Redlock from 'redlock';

const redis = new Redis({
  host: 'redis-cluster.internal',
  port: 6379,
  // TLS, auth, etc.
});

const redlock = new Redlock([redis], {
  driftFactor: 0.01,
  retryCount: 32,
  retryDelay: 100,  // ms
  retryJitter: 200, // ms
  automaticExtensionThreshold: 500, // ms
});

export async function withLock(name, fn, ttlMs = 5000) {
  const lock = await redlock.acquire([`vibe:${name}`], ttlMs);
  try {
    return await fn();
  } finally {
    await lock.release();
  }
}

Use case: Vibe Coding OS chạy trên K8s, nhiều pod. Cần Redis cluster (3+ nodes) cho quorum.

2.4 Giải pháp C: PostgreSQL advisory lock (production-grade)

// runtime/core/pg-store.mjs (mới)
import { Pool } from 'pg';

const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  // max: 20, idleTimeoutMillis: 30000, ...
});

export async function withLock(store, name, fn) {
  const client = await pool.connect();
  try {
    // Hash name thành bigint
    const lockKey = hashStringToBigInt(name);
    await client.query('BEGIN');
    await client.query('SELECT pg_advisory_xact_lock($1)', [lockKey]);
    const result = await fn();
    await client.query('COMMIT');
    return result;
  } catch (err) {
    await client.query('ROLLBACK');
    throw err;
  } finally {
    client.release();  // tự unlock khi transaction kết thúc
  }
}

function hashStringToBigInt(s) {
  return parseInt(createHash('sha256').update(s).digest('hex').slice(0, 16), 16);
}

3 Sandbox OS-level

Vấn đề: repo chỉ "instruction contract", không ngăn code chạy ngoài ý muốn.

3.1 Giải pháp A: Docker (phổ biến nhất)

Wrap mọi exec call trong Docker container với resource limits:

# runtime/sandbox/Dockerfile
FROM node:20-slim

# Drop privileges ngay
RUN useradd -m -u 1001 -s /bin/bash sandbox \
    && mkdir -p /workspace /runtime \
    && chown -R sandbox:sandbox /workspace /runtime

RUN apt-get update && apt-get install -y --no-install-recommends \
    tini \
    && rm -rf /var/lib/apt/lists/*

USER sandbox
WORKDIR /workspace
ENTRYPOINT ["/usr/bin/tini", "--"]
// runtime/sandbox/sandbox-entry.mjs
import { exec } from 'node:child_process';
import { promisify } from 'node:util';

const execAsync = promisify(exec);

const ALLOWED_COMMANDS = new Set([
  'ls', 'cat', 'grep', 'find', 'head', 'tail',
  'node', 'npm', 'npx', 'git',
  'echo', 'pwd', 'wc', 'sort', 'uniq',
]);

const ALLOWED_READ_PREFIXES = ['/workspace/', '/runtime/'];

process.on('message', async ({ id, command, cwd }) => {
  const [bin, ...args] = command.split(/\s+/);

  // Whitelist check
  if (!ALLOWED_COMMANDS.has(bin)) {
    return process.send({
      id, error: `Command not allowed: ${bin}`,
      allowed: [...ALLOWED_COMMANDS],
    });
  }

  // Path traversal check
  for (const arg of args) {
    if (arg.startsWith('/') &&
        !ALLOWED_READ_PREFIXES.some(p => arg.startsWith(p))) {
      return process.send({
        id, error: `Path not allowed: ${arg}`,
        allowed: ALLOWED_READ_PREFIXES,
      });
    }
  }

  try {
    const { stdout, stderr } = await execAsync(command, {
      cwd: resolve('/workspace', cwd || '.'),
      timeout: 30_000,
      maxBuffer: 10 * 1024 * 1024,
      env: { PATH: '/usr/local/bin:/usr/bin:/bin' },  // KHÔNG inherit
    });
    process.send({ id, stdout, stderr });
  } catch (err) {
    process.send({ id, error: err.message });
  }
});
// scripts/sandbox-exec.mjs (entry từ Vibe Coding OS)
import { spawn } from 'node:child_process';

export async function sandboxExec(command, options = {}) {
  return new Promise((resolve, reject) => {
    const docker = spawn('docker', [
      'run', '--rm', '-i',
      '--cpus=0.5',                      // giới hạn CPU
      '--memory=512m',                   // giới hạn RAM
      '--pids-limit=100',                // giới hạn process
      '--read-only',                     // root FS read-only
      '--tmpfs', '/tmp:size=100m',       // tmp có giới hạn
      '--network=none',                  // tắt network
      '-v', `${process.cwd()}:/workspace:ro`,  // read-only mount
      '-v', `${process.env.HOME}/.vibe-sandbox:/runtime:rw`,
      'vibe-coding-os-sandbox:latest',
      'node', '/runtime/sandbox-entry.mjs',
    ], { stdio: ['pipe', 'pipe', 'pipe'] });

    docker.stdin.write(JSON.stringify({ id: 1, command }) + '\n');
    docker.stdin.end();

    let output = '';
    docker.stdout.on('data', (chunk) => output += chunk);
    docker.on('close', (code) => {
      if (code !== 0) reject(new Error(`sandbox exit ${code}`));
      else resolve(JSON.parse(output));
    });
  });
}

// scripts/firejail-exec.mjs
import { spawn } from 'node:child_process';

export async function firejailExec(command) {
  return new Promise((resolve, reject) => {
    const proc = spawn('firejail', [
      '--profile=/etc/firejail/vibe-coding-os.profile',
      '--',
      '/bin/bash', '-c', command,
    ], { stdio: 'pipe' });

    let out = '', err = '';
    proc.stdout.on('data', (d) => out += d);
    proc.stderr.on('data', (d) => err += d);
    proc.on('close', (code) => {
      if (code === 0 || code === 1) resolve({ stdout: out, stderr: err });
      else reject(new Error(`firejail exit ${code}: ${err}`));
    });
  });
}

3.3 Giải pháp C: E2B (cloud sandbox cho AI agent)

// scripts/e2b-exec.mjs
import { Sandbox } from '@e2b/sdk';

export async function e2bExec(command, options = {}) {
  // E2B cung cấp ephemeral Linux sandbox qua cloud
  const sandbox = await Sandbox.create({
    template: 'node-20',
    timeoutMs: options.timeoutMs || 5 * 60_000,  // 5 min
  });

  try {
    // Tất cả execution xảy ra trong E2B cloud, không chạm máy bạn
    const result = await sandbox.process.startAndWait(
      `cd /workspace && ${command}`,
      { timeoutMs: 30_000 }
    );
    return {
      stdout: result.stdout,
      stderr: result.stderr,
      exitCode: result.exitCode,
    };
  } finally {
    await sandbox.kill();
  }
}

Use case: Khi AI agent cần chạy code lạ (pull từ GitHub random, chạy example, test thư viện mới). E2B cung cấp "throwaway Linux VM" qua API, tự hủy sau khi xong.

4.2 Bước 1: Implement supermemory adapter

// adapters/memory/supermemory/index.mjs
import { createClient } from '@supermemory/sdk';

const client = createClient({
  apiKey: process.env.SUPERMEMORY_API_KEY,
});

export const supermemoryAdapter = {
  name: 'supermemory',

  // === REQUIRED ===
  async search(query, options = {}) {
    const result = await client.search({
      q: query,
      limit: options.limit || 10,
      containerTags: ['vibe-coding-os'],
    });
    return result.results.map(r => ({
      id: r.id,
      content: r.content,
      score: r.score,
      tags: r.tags,
      createdAt: r.createdAt,
    }));
  },

  async ingest(item) {
    // Privacy: redact TRƯỚC khi gửi cloud
    const redactedContent = redactText(item.content);

    return await client.add({
      content: redactedContent,
      type: item.kind || 'observation',
      containerTag: 'vibe-coding-os',
      metadata: { source: item.source, scope: item.scope },
    });
  },

  // === OPTIONAL ===
  async delete(id) { return await client.delete(id); },
  async update(id, patch) { return await client.update(id, patch); },

  // === HEALTH CHECK ===
  async healthCheck() {
    try {
      await client.ping();
      return { ok: true };
    } catch (err) {
      return { ok: false, error: err.message, fallback: 'local-fs' };
    }
  },

  capabilities: {
    vectorSearch: true,
    fullTextSearch: true,
    asyncIngest: true,
    privacyRedaction: false,  // chúng ta redact trước khi gửi
  },
};

          source: args.source || 'runtime-mcp',
          tags: args.tags,

5 AI tự quyết (triết lý)

Đây không phải limitation, đây là design choice.

5.1 Tại sao Vibe Coding OS chọn Human-in-loop

CONSTITUTION.md, principle 1

Human intent stays sovereign. The assistant proposes; it does not invent requirements or silently expand scope.

5.2 Lý do thực tế

1. Hallucination

AI tin nó đúng khi nó sai. AI tự quyết scope → scope creep, mất thời gian, mất trust.

2. Blast radius

AI tự quyết destructive action (rm -rf, force push, npm publish) → mất dữ liệu, mất công sức.

3. Accountability

Khi AI sai, ai chịu trách nhiệm? Human-in-loop: con người chịu, vì con người approve cuối cùng.

4. Trust calibration

Approve mỗi lần = cơ hội học convention. Skip approval = mất cơ hội align AI với codebase.

5.3 Nếu vẫn cần AI tự quyết (chấp nhận rủi ro)

Đây là pattern Human-on-the-loop thay vì Human-in-the-loop:

// runtime/core/auto-approve.mjs
export function createAutoApprover(policy) {
  // Auto-approve khi:
  //   - Risk level = 'safe' hoặc 'review' VÀ
  //   - Action nằm trong whitelist VÀ
  //   - KHÔNG phải destructive command

  const DESTRUCTIVE_ACTIONS = new Set([
    'git push', 'npm publish', 'rm', 'rmdir', 'drop table',
    'kubectl delete', 'terraform destroy', 'docker system prune',
    'release.publish',
  ]);

  return {
    shouldAutoApprove(action, risk) {
      if (risk?.level === 'dangerous' || risk?.level === 'blocked') return false;
      if (DESTRUCTIVE_ACTIONS.has(action)) return false;
      return policy.autoApproveActions.has(action);
    },

    async handle(action, context) {
      if (this.shouldAutoApprove(action, context.risk)) {
        // Log vàng — chỉ audit, không dừng
        await logAutoApproval(action, context);
        return { status: 'auto-approved', actor: 'auto-approver' };
      }
      // Ngược lại: yêu cầu human
      return await createPendingApproval(action, context);
    },
  };
}

const policy = {
  autoApproveActions: new Set([
    'file.read', 'file.write',    // file ops trong workspace
    'npm.test', 'npm.lint',       // local validation
    'git.add', 'git.commit',      // local commit
    // KHÔNG auto-approve: git push, npm publish, rm
  ]),
};

5.5 Bật auto-mode (Vibe Coding OS đã có sẵn gate)

// runtime/mcp/server.mjs
import { withApprovalGate } from '../core/approval-gate.mjs';

const AUTO_APPROVE = process.env.VIBE_AUTO_APPROVE === 'true';
const DESTRUCTIVE = new Set(['rm', 'rmdir', 'git push', 'npm publish']);

export function startServer() {
  for (const tool of buildTools(store)) {
    server.setRequestHandler(CallToolRequestSchema, async (request) => {
      const handler = tool.handler;

      if (AUTO_APPROVE && !DESTRUCTIVE.has(tool.name)) {
        // Skip approval gate cho non-destructive
        return await handler(request.params.arguments);
      }

      // Mặc định: qua gate
      return await withApprovalGate(handler, store, tool.name)(
        request.params.arguments
      );
    });
  }
}
Khuyến nghị

Bắt đầu với default. Khi bạn đã tin tưởng AI với task X, thêm X vào autoApproveActions. Luôn giữ destructive actions ngoài whitelist.

6.5 Tóm tắt: Vibe Coding OS ở layer nào?

Vai trò của Vibe Coding OS

Vibe Coding OS chiếm layer 2-3 (dev-time). Layer 1 (editor), layer 4 (CI), layer 5 (platform) thuộc về editor và GitHub platform. Đây là separation of concerns tốt: mỗi tool làm tốt một việc.

7 Decision Tree cuối cùng

"Tôi có nên dùng Vibe Coding OS không? Khi nào cần kết hợp gì?"

1
Bạn cần spec-driven workflow + spec/plan/tasks trước code?
→ Có: Dùng Vibe Coding OS. Đây là core value.
2
Bạn cần AI nhớ convention qua session?
→ Có: Dùng Vibe Coding OS memory layer (20 skills).
3
Bạn cần chạy trên nhiều AI editor (Claude Code + Cursor + Codex)?
→ Có: Dùng Vibe Coding OS. 9 adapters.
4
Bạn cần defense chống prompt injection + secret leak?
→ Có: Dùng Vibe Coding OS. 3-layer security.
5
Bạn cần distributed lock (multi-machine)?
→ Có: Vibe Coding OS không đủ. Thêm SQLite WAL / Redis / Postgres.
6
Bạn cần sandbox OS-level (chạy code lạ an toàn)?
→ Có: Vibe Coding OS không đủ. Thêm Docker / firejail / E2B.
7
Bạn cần hosted memory (multi-device sync)?
→ Có: Vibe Coding OS có adapter. Wire supermemory / Qdrant / pgvector.
8
Bạn cần AI tự quyết hoàn toàn (no human)?
→ Có: Vibe Coding OS chống pattern này theo triết lý. Hoặc dùng tool khác (AutoGPT, MetaGPT), hoặc bật auto-approve với whitelist chặt.
9
Bạn cần branch protection / secret scanner chuyên dụng?
→ Có: Bắt buộc dùng GitHub Platform + gitleaks + branch protection. Vibe Coding OS bổ trợ, không thay thế.
